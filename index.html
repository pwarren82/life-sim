async function callImagenAPI(promptText, caption) { setImgStatus("Rendering character visual via Imagen 4..."); const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${state.apiKey}`; try { const res = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-goog-api-key': state.apiKey }, body: JSON.stringify({ instances: [{ prompt: promptText }], parameters: { sampleCount: 1, aspectRatio: "1:1" } }) }); const data = await res.json(); if (data.error) throw new Error(data.error.message); if (data.predictions && data.predictions[0] && data.predictions[0].bytesBase64Encoded) { const base64 = data.predictions[0].bytesBase64Encoded; state.lastImage = base64; state.lastCaption = caption; localStorage.setItem('life_sim_image', base64); localStorage.setItem('life_sim_caption', caption); displayImage(base64, caption); setImgStatus(""); } else { throw new Error("No image data returned from API."); } } catch (err) { setImgStatus(`Image API Notice: ${err.message}`, true); } }
